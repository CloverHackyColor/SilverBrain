# Precompile individual Metal kernels for ggml-metal.
if(APPLE AND GGML_METAL_EMBED_LIBRARY AND TARGET ggml-metal)

    # The generated .metal files are created by ggml-metal's ASM commands.
    get_target_property(GGML_METAL_BINARY_DIR ggml-metal BINARY_DIR)
    set(GGML_METAL_AUTOGEN_DIR
        "${GGML_METAL_BINARY_DIR}/autogenerated"
    )

    # Keep this list in exactly the same order/names as k_lib_names[].
    set(SILVERBRAIN_METAL_KERNELS
        fa_aux
        fa_f16
        fa_f32
        fa_q4_0
        fa_q4_1
        fa_q5_0
        fa_q5_1
        fa_q8_0
        fa_tensor
        fa_vec_f16
        fa_vec_f32
        fa_vec_q4_0
        fa_vec_q4_1
        fa_vec_q5_0
        fa_vec_q5_1
        fa_vec_q8_0
        mul_mv
        mul_mv_mma
        mul_mm
        quantize
        softmax
        norm
        unary
        binbcast
        reduce
        tri
        ssm
        wkv
        gated_delta_net
        solve_tri
        rope
        conv
        upscale
        argsort
        pool
        misc
    )

    set(SILVERBRAIN_METAL_OUTPUT_DIR
        "${CMAKE_CURRENT_BINARY_DIR}/silverbrain-metal-kernels"
    )

    file(MAKE_DIRECTORY "${SILVERBRAIN_METAL_OUTPUT_DIR}")

    # Match the Metal compiler flags used by ggml-metal.
    set(SILVERBRAIN_METAL_SDK "${CMAKE_OSX_SYSROOT}")
    if(NOT SILVERBRAIN_METAL_SDK)
        set(SILVERBRAIN_METAL_SDK macosx)
    endif()

    if(GGML_METAL_SHADER_DEBUG)
        set(SILVERBRAIN_METAL_XC_FLAGS
            -fno-fast-math
            -fno-inline
        )
    else()
        set(SILVERBRAIN_METAL_XC_FLAGS -O3)
    endif()

    if(CMAKE_OSX_SYSROOT MATCHES "[Ss]imulator")
        set(SILVERBRAIN_METAL_TARGET_SIM "-simulator")
    else()
        set(SILVERBRAIN_METAL_TARGET_SIM "")
    endif()

    if(GGML_METAL_MACOSX_VERSION_MIN)
        list(APPEND SILVERBRAIN_METAL_XC_FLAGS
            "-mmacosx-version-min=${GGML_METAL_MACOSX_VERSION_MIN}"
        )
    elseif(NOT GGML_METAL_TARGET_OS STREQUAL "macos"
           AND CMAKE_OSX_DEPLOYMENT_TARGET)
        list(APPEND SILVERBRAIN_METAL_XC_FLAGS
            "-mtargetos=${GGML_METAL_TARGET_OS}${CMAKE_OSX_DEPLOYMENT_TARGET}${SILVERBRAIN_METAL_TARGET_SIM}"
        )
    endif()

    if(GGML_METAL_STD)
        list(APPEND SILVERBRAIN_METAL_XC_FLAGS
            "-std=${GGML_METAL_STD}"
        )
    endif()

    set(SILVERBRAIN_METAL_LIBRARIES "")
    set(SILVERBRAIN_METAL_EMBED_ASM_FILES "")

    foreach(kind IN LISTS SILVERBRAIN_METAL_KERNELS)

        set(embed_metal
            "${GGML_METAL_AUTOGEN_DIR}/ggml-metal-embed-${kind}.metal"
        )

        set(embed_asm
            "${GGML_METAL_AUTOGEN_DIR}/ggml-metal-embed-${kind}.s"
        )

        set(air
            "${SILVERBRAIN_METAL_OUTPUT_DIR}/${kind}.air"
        )

        set(metallib
            "${SILVERBRAIN_METAL_OUTPUT_DIR}/${kind}.metallib"
        )

        list(APPEND SILVERBRAIN_METAL_LIBRARIES "${metallib}")
        list(APPEND SILVERBRAIN_METAL_EMBED_ASM_FILES "${embed_asm}")

        add_custom_command(
            OUTPUT "${metallib}"

            COMMAND xcrun -sdk "${SILVERBRAIN_METAL_SDK}"
                    metal ${SILVERBRAIN_METAL_XC_FLAGS}
                    -I "${GGML_METAL_INCLUDE_DIR}"
                    -c "${embed_metal}" -o "${air}"

            COMMAND xcrun -sdk "${SILVERBRAIN_METAL_SDK}"
                    metallib "${air}" -o "${metallib}"

            DEPENDS "${embed_asm}"

            COMMENT "Precompiling Metal kernel ${kind}"
            VERBATIM
        )

    endforeach()

    # Generate the same per-kernel SHA-256 values that the runtime loader
    # calculates from the embedded source bytes.
    set(SILVERBRAIN_METAL_FINGERPRINT
        "${SILVERBRAIN_METAL_OUTPUT_DIR}/fingerprint"
    )

    set(SILVERBRAIN_METAL_FINGERPRINT_SCRIPT
        "${CMAKE_CURRENT_BINARY_DIR}/write-silverbrain-metal-fingerprint.cmake"
    )

    file(WRITE "${SILVERBRAIN_METAL_FINGERPRINT_SCRIPT}" [=[
        if(NOT DEFINED SOURCE_DIR OR NOT DEFINED OUTPUT_FILE OR NOT DEFINED KERNEL_NAMES)
            message(FATAL_ERROR "Missing fingerprint script arguments")
        endif()

        set(content "")

        foreach(kind IN LISTS KERNEL_NAMES)
            set(source "${SOURCE_DIR}/ggml-metal-embed-${kind}.metal")

            if(NOT EXISTS "${source}")
                message(FATAL_ERROR "Missing generated Metal source: ${source}")
            endif()

            file(SHA256 "${source}" hash)
            string(APPEND content "${kind} ${hash}\n")
        endforeach()

        file(WRITE "${OUTPUT_FILE}" "${content}")
        ]=])

    add_custom_command(
        OUTPUT "${SILVERBRAIN_METAL_FINGERPRINT}"

        COMMAND "${CMAKE_COMMAND}"
                "-DSOURCE_DIR=${GGML_METAL_AUTOGEN_DIR}"
                "-DOUTPUT_FILE=${SILVERBRAIN_METAL_FINGERPRINT}"
                "-DKERNEL_NAMES=${SILVERBRAIN_METAL_KERNELS}"
                -P "${SILVERBRAIN_METAL_FINGERPRINT_SCRIPT}"

        DEPENDS ${SILVERBRAIN_METAL_EMBED_ASM_FILES}

        COMMENT "Generating SilverBrain Metal kernel fingerprint"
        VERBATIM
    )

    add_custom_target(silverbrain-metal-kernels ALL
        DEPENDS
            ${SILVERBRAIN_METAL_LIBRARIES}
            "${SILVERBRAIN_METAL_FINGERPRINT}"
    )

    # The generated embedded sources must exist before precompilation.
    add_dependencies(silverbrain-metal-kernels ggml-metal)

    # Build the precompiled kernels before building the application.
    add_dependencies(SilverBrain silverbrain-metal-kernels)

endif()
